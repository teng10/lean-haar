\subsection{Commutant of the First Order Unitary Group}

\begin{remark}[Schur's Lemma]\label{rm:SchursLemma}
    For finite-dimensional complex representation $V$ of a group $G$ that is irreducible under the action of $\mathcal{U}(2)$, every $G$-equivariant endormorphism of $V$ (i.e. linear map from $V \rightarrow V$) is a scalar multiple of the identity.
\end{remark}

\begin{proposition}\label{lemma:Transitivity}
Let $v, w \in \mathbb{C}^2$ be unit vectors (i.e., $\|v\| = \|w\| = 1$). Then there exists a unitary matrix $U \in U(2)$ such that $Uv = w$.
\end{proposition}

\begin{proof}
Let $S^3 = \{ z \in \mathbb{C}^2 : \|z\| = 1 \}$ denote the unit sphere in $\mathbb{C}^2$, which is acted upon by $\mathcal{U}(2)$ by matrix multiplication.
Since unitary matrices preserve the Hermitian inner product, for any $U \in \mathcal{U}(2)$ and vector $z \in \mathbb{C}^2$,
\begin{equation}
  |Uz|^2 = \langle Uz, Uz \rangle = \langle z, U^{\dagger} U z \rangle = \langle z, z \rangle = |z|^2.
\end{equation}
Hence the action of $U$ is restricted to a well-defined action on $S^3$.

We show that the action of $\mathcal{U}(2)$ on $S^3$ is transitive.
That is, for any $v, w \in S^3$, the orbit of $v$ contains $w$.
Let $e_1$ be the standard basis vector in $S^{3}$, where $e_1$ is given by
\begin{equation}
e_1 = \begin{pmatrix} 1 \\ 0 \end{pmatrix}.
\end{equation}
Let $x \in S^3$ be an arbitrary unit vector. We can extend the singleton set $\{x\}$ to an orthonormal basis $\mathcal{B}_x = \{x, x^\perp\}$ of $\mathbb{C}^2$. Similarly, let $\mathcal{B}_{e_1} = \{e_1, e_2\}$ be the standard orthonormal basis.

Define a linear operator $T: \mathbb{C}^2 \to \mathbb{C}^2$ by specifying its action on the basis $\mathcal{B}_{e_1}$:
\begin{equation}
T(e_1) = x \quad \text{and} \quad T(e_2) = x^\perp.
\end{equation}
Since $T$ maps an orthonormal basis to an orthonormal basis, it preserves inner products. Specifically, for any $u, v \in \mathbb{C}^2$, $\langle Tu, Tv \rangle = \langle u, v \rangle$. A linear operator on a finite-dimensional inner product space that preserves the inner product is unitary. Thus, $T \in U(2)$. By construction, $T e_1 = x$.

Since $x$ was arbitrary, the orbit of $e_1$ is $S^3$. Now, let $v, w \in S^3$ be the given vectors. By the transitivity established above, there exist matrices $U_1, U_2 \in U(2)$ such that:
\begin{equation}
U_1 e_1 = v \quad \text{and} \quad U_2 e_1 = w.
\begin{equation}
Since $U(2)$ is a group, $U_1^{-1}$ exists and is unitary. Define $U = U_2 U_1^{-1}$. Then $U \in U(2)$, and:
\begin{equation}
U v = (U_2 U_1^{-1}) v = U_2 (U_1^{-1} v) = U_2 e_1 = w.
\end{equation}
Thus, there exists a unitary matrix $U$ such that $Uv = w$.
\end{proof}

\begin{theorem}[Abstract: Commutant of first order unitaries is a scalar multiple of the identity]
    Given the identity $\mathbb{I} \in \mathcal{U}(2)$,
    \begin{equation}
        \{M : M U = U M, \forall U \in \mathcal{U}(2) \} = \{\lambda \mathbb{I} : \lambda \in \mathbb{C} \}.
    \end{equation}
\end{theorem}

\begin{proof}
We first show forward inclusion, where $\{M : M U = U M, \forall U \in \mathcal{U}(2) \} \subseteq \{\lambda \mathbb{I} : \lambda \in \mathbb{C} \}$.
Let $V = \mathbb{C}^2$ be the complex vector space of dimension $2$ acted upon by $\mathcal{U}(2)$.
We show that $V$ is irreducible.

Clearly $\{0\}$ is a subspace of $V$.
Suppose $W \subseteq V$ is a nonzero $\mathcal{U}$(2)-invariant subspace.
Note that the natural action of $\mathcal{U}(2)$ is transitive on the unit sphere of $\mathbb{C}^2$, as given in Lemma~\ref{lemma:Transitivity}.
Since $V$ has dimension $2$ by definition, $\text{dim}(W) = 1$ or $\text{dim}(W) = 2$.
Assume for contradiction that $\text{dim}(W) = 1$, and let $v \in W$ be a unit vector.
By transitivity, $\mathcal{U}(2)v$ covers the entire unit sphere of $V$.
That is, for all $w \in W$, there exists a $U$ such that $Uv = w$.
Since $W$ is $\mathcal{U}(2)$-invariant, $W$ must contain the entire unit sphere and therefore spans $V$, implying $\text{dim}(W) = 2$.
So $V = W$.
Since the only subspaces of $V$ are the zero subspace and itself, $V$ is irreducible.
Note that $MU = UM$ implies $M$ is a $\mathcal{U}(2)$-equivariant endomorphism of $V$.
By Schur's lemma, as restated in Remark~\ref{rm:SchursLemma}
\begin{equation}
    M = \lambda \mathbb{I},
\end{equation}
where $\lambda \in \mathbb{C}$.

Conversely, it is trivial that any scalar multiple of the identity commutes with all $U \in \mathcal{U}(2)$.
\end{proof}

We now write an outline based on the Schur-Weyl duality.

\begin{lemma}[Span of 1D permutations is the span of $\mathbb{I}$]\label{lemma:1DSpanPermutationEqId}
    Let $d \in \mathbb{N}$ and $k = 1$.
    Then
    \begin{equation}
        \text{Span}\{ V_{d}(\pi) : \pi \in S_{k} \} = \text{Span}(\mathbb{I}).
    \end{equation}
\end{lemma}

\begin{proof}
    In a tensor product composed of one element, any permutation of the tensor product simply returns the element itself.
\end{proof}

\begin{theorem}[Using Schur-Weyl Duality for Commutants]
    Given the identity $\mathbb{I} \in \mathcal{U}(2)$,
    \begin{equation}
        \{M : M U = U M, \forall U \in \mathcal{U}(2) \} = \{\lambda \mathbb{I} : \lambda \in \mathbb{C} \}.
    \end{equation}
    We prove this statement now using the abstract Schur-Weyl duality.
\end{theorem}

\begin{proof}
    Observe that the left-hand side can be equivalently written as
    \begin{equation}
        \{M : MU = UM, \forall U \in \mathcal{U}(2) \} = \{M : MU^{\otimes 1} = U^{\otimes 1} M, \forall U \in \mathcal{U}(2) \}.
    \end{equation}
    By the Schur-Weyl duality and transitivity of equality,
    \begin{equation}
        \{M : MU = UM, \forall U \in \mathcal{U}(2) \} = \text{Span}(\mathbb{I}) = \{ \lambda \mathbb{I} : \lambda \in \mathbb{C} \}
    \end{equation}
\end{proof}

\begin{theorem}[Assume Tensor Power of 1 is Known]
    Given the identity $\mathbb{I} \in \mathcal{U}(2)$,
    \begin{equation}
        \{M : M U^{\otimes 1} = U^{\otimes 1} M, \forall U \in \mathcal{U}(2) \} = \{\lambda \mathbb{I} : \lambda \in \mathbb{C} \}.
    \end{equation}
    We prove this statement now using the abstract Schur-Weyl duality.
\end{theorem}
