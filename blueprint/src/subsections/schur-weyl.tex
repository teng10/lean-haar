\section{Definitions and Commutants}

Let $V = \mathbb{C}^d$ denote the $d$-dimensional complex Euclidean space (modeled as \texttt{Fin d $\to$ $\mathbb{C}$}). We study the $k$-fold tensor power of $V$ and the actions of the symmetric group $S_k$ and the endomorphism algebra $\text{End}(V)$ on it.

\begin{definition}[Tensor Power Space]\label{def:TensV}
  \lean{SchurWeyl.TensV}\leanok
  The $k$-fold tensor power of $V$ is defined as:
  \begin{equation}
    V^{\otimes k} = \bigotimes_{i=1}^k V
  \end{equation}
\end{definition}

\begin{definition}[Permutation Action]\label{def:permAction}
  \lean{SchurWeyl.permAction}\uses{def:TensV}\leanok
  Given a permutation $\sigma \in S_k$, the permutation operator $W_\sigma$ on $V^{\otimes k}$ acts by permuting the tensor factors:
  \begin{equation}
    W_\sigma(v_1 \otimes \dots \otimes v_k) = v_{\sigma^{-1}(1)} \otimes \dots \otimes v_{\sigma^{-1}(k)}
  \end{equation}
\end{definition}

\begin{definition}[Diagonal Action]\label{def:diagAction}
  \lean{SchurWeyl.diagAction}\uses{def:TensV}\leanok
  Given an operator $g \in \text{End}(V)$, the diagonal action $g^{\otimes k}$ on $V^{\otimes k}$ acts as $g$ on each tensor factor:
  \begin{equation}
    g^{\otimes k}(v_1 \otimes \dots \otimes v_k) = g(v_1) \otimes \dots \otimes g(v_k)
  \end{equation}
\end{definition}

\begin{definition}[Permutation Image]\label{def:permImage}
  \lean{SchurWeyl.permImage}\uses{def:permAction, def:TensV}\leanok
  The permutation image $\mathcal{P}_{d,k} \subseteq \text{End}(V^{\otimes k})$ is the set of all permutation operators:
  \begin{equation}
    \mathcal{P}_{d,k} = \{ W_\sigma \mid \sigma \in S_k \}
  \end{equation}
\end{definition}

\begin{definition}[Diagonal Image]\label{def:diagImage}
  \lean{SchurWeyl.diagImage}\uses{def:diagAction, def:TensV}\leanok
  The diagonal image $\mathcal{D}_{d,k} \subseteq \text{End}(V^{\otimes k})$ is the set of all diagonal action operators:
  \begin{equation}
    \mathcal{D}_{d,k} = \{ g^{\otimes k} \mid g \in \text{End}(V) \}
  \end{equation}
\end{definition}

\begin{lemma}[Permutation Action on Simple Tensors]\label{lem:permAction_tprod}
  \lean{SchurWeyl.permAction_tprod}\uses{def:permAction}\leanok
  For any simple tensor $\bigotimes_{i} v_i$, we have:
  \begin{equation}
    W_\sigma(\bigotimes_{i} v_i) = \bigotimes_{i} v_{\sigma^{-1}(i)}
  \end{equation}
\end{lemma}

\begin{lemma}[Diagonal Action on Simple Tensors]\label{lem:diagAction_tprod}
  \lean{SchurWeyl.diagAction_tprod}\uses{def:diagAction}\leanok
  For any simple tensor $\bigotimes_{i} v_i$, we have:
  \begin{equation}
    g^{\otimes k}(\bigotimes_{i} v_i) = \bigotimes_{i} g(v_i)
  \end{equation}
\end{lemma}

\begin{lemma}[Action Commutativity on Simple Tensors]\label{lem:permAction_diagAction_tprod}
  \lean{SchurWeyl.permAction_diagAction_tprod}\uses{lem:permAction_tprod, lem:diagAction_tprod, def:diagAction, def:permAction}\leanok
  For any permutation $\sigma \in S_k$, operator $g \in \text{End}(V)$, and vector family $v$:
  \begin{equation}
    W_\sigma(g^{\otimes k}(v_1 \otimes \dots \otimes v_k)) = g^{\otimes k}(W_\sigma(v_1 \otimes \dots \otimes v_k))
  \end{equation}
\end{lemma}

\begin{theorem}[Permutation-Diagonal Commutativity]\label{thm:permAction_diagAction_comm}
  \lean{SchurWeyl.permAction_diagAction_comm}\uses{lem:permAction_diagAction_tprod, def:diagAction, def:permAction}\leanok
  For any $\sigma \in S_k$ and $g \in \text{End}(V)$, the permutation action and diagonal action commute as endomorphisms:
  \begin{equation}
    W_\sigma \circ g^{\otimes k} = g^{\otimes k} \circ W_\sigma
  \end{equation}
\end{theorem}

\begin{theorem}[Permutation Image Centralizes Diagonal Image]\label{thm:permImage_subset_centralizer_diagImage}
  \lean{SchurWeyl.permImage_subset_centralizer_diagImage}\uses{thm:permAction_diagAction_comm, def:permImage, def:diagImage}\leanok
  The permutation image is contained in the centralizer of the diagonal image:
  \begin{equation}
    \mathcal{P}_{d,k} \subseteq \text{Comm}(\mathcal{D}_{d,k})
  \end{equation}
\end{theorem}

\begin{theorem}[Diagonal Image Centralizes Permutation Image]\label{thm:diagImage_subset_centralizer_permImage}
  \lean{SchurWeyl.diagImage_subset_centralizer_permImage}\uses{thm:permAction_diagAction_comm, def:permImage, def:diagImage}\leanok
  The diagonal image is contained in the centralizer of the permutation image:
  \begin{equation}
    \mathcal{D}_{d,k} \subseteq \text{Comm}(\mathcal{P}_{d,k})
  \end{equation}
\end{theorem}


\section{First Fundamental Theorem of Invariant Theory}

To prove the hard direction of Schur-Weyl duality, we first prove that the centralizer of the permutation image is contained in the span of the diagonal image. This is a form of the First Fundamental Theorem (FFT) of invariant theory.

\begin{definition}[Tensor Basis]\label{def:tensorBasis}
  \lean{SchurWeyl.tensorBasis}\uses{def:TensV}\leanok
  The standard basis of $V^{\otimes k}$ constructed as the tensor product of the standard basis of $V$:
  \begin{equation}
    (e_I)_{I \in [d]^k}
  \end{equation}
\end{definition}

\begin{definition}[Endomorphism Matrix Representation]\label{def:toEndMatrix}
  \lean{SchurWeyl.toEndMatrix}\uses{def:tensorBasis, def:TensV}\leanok
  The linear equivalence between the endomorphism algebra $\text{End}(V^{\otimes k})$ and the algebra of $d^k \times d^k$ complex matrices:
  \begin{equation}
    \text{End}(V^{\otimes k}) \simeq \text{Mat}_{d^k \times d^k}(\mathbb{C})
  \end{equation}
\end{definition}

\begin{lemma}[Permutation Action on Tensor Basis]\label{lem:permAction_tensorBasis}
  \lean{SchurWeyl.permAction_tensorBasis}\uses{def:tensorBasis, def:permAction}\leanok
  The permutation action of $W_\sigma$ on a basis element $e_I$ permutes the indices:
  \begin{equation}
    W_\sigma(e_I) = e_{I \circ \sigma^{-1}}
  \end{equation}
\end{lemma}

\begin{lemma}[Matrix of Permutation Action]\label{lem:toEndMatrix_permAction}
  \lean{SchurWeyl.toEndMatrix_permAction}\uses{def:toEndMatrix, lem:permAction_tensorBasis, def:permAction}\leanok
  The matrix entries of the permutation action $W_\sigma$ are:
  \begin{equation}
    (W_\sigma)_{I, J} = \delta_{I, J \circ \sigma^{-1}}
  \end{equation}
\end{lemma}

\begin{lemma}[Matrix Representation of Diagonal Action]\label{lem:toEndMatrix_diagAction}
  \lean{SchurWeyl.toEndMatrix_diagAction}\uses{def:toEndMatrix, def:diagAction, def:tensorBasis}\leanok
  For any operator $g \in \text{End}(V)$, the matrix elements of its diagonal action $g^{\otimes k}$ in the tensor basis are given by:
  \begin{equation}
    (g^{\otimes k})_{I, J} = \prod_{m=1}^k g_{I(m), J(m)}
  \end{equation}
\end{lemma}

\begin{lemma}[DiagAction of Diagonal Operator]\label{lem:diagAction_diagonal}
  \lean{SchurWeyl.diagAction_diagonal}\uses{def:toEndMatrix, def:diagAction,lem:toEndMatrix_diagAction}\leanok
  If $g \in \text{End}(V)$ is a diagonal operator with entries $f$, then $g^{\otimes k}$ is a diagonal operator with entries $\prod_{m=1}^k f(I(m))$.
\end{lemma}

\begin{lemma}[Monomial Coefficient Independence]\label{lem:monomial_coeff_zero_of_eval_zero}
  \lean{monomial_coeff_zero_of_eval_zero}\leanok
  If a linear combination of distinct monomials in $d^2$ variables over $\mathbb{C}$ evaluates to zero at all points, then all coefficients are zero.
\end{lemma}

{\chat
The proof groups pairs of index functions $I,J:[k]\to[d]$ by their
pair multiplicities
\[
  c_{I,J}(a,b)=\#\{m\mid (I(m),J(m))=(a,b)\}.
\]
These multiplicities classify the simultaneous permutation orbits: two pairs
have the same multiplicities if and only if one is obtained from the other by
precomposing both indices with a permutation. Hence, if $X$ commutes with all
$W_\sigma$, its entries $X_{I,J}$ are constant on each such class. The product
in the diagonal-action matrix formula is the corresponding monomial:
\[
  \prod_{m=1}^k g_{I(m),J(m)}
    =\prod_{(a,b)\in[d]^2}g_{a,b}^{c_{I,J}(a,b)}.
\]
The pair-count definition, orbit classification and regrouping lemmas are
private auxiliaries in the Lean proof.

Suppose that $X$ does not belong to $\text{Span}(\mathcal{D}_{d,k})$.
Mathlib's separation theorem gives a linear
functional $\varphi$ vanishing on this span with $\varphi(X)\ne0$.
Write $\varphi(T)=\sum_{I,J}C_{I,J}T_{I,J}$. Vanishing on every

\section{Double Commutant Theorem and Schur-Weyl Duality}

{\chat
Let $A=\text{Span}(\mathcal{P}_{d,k})$. To prove the double commutant
inclusion, we use the permutation action to regard $M=V^{\otimes k}$ as a
$\mathbb{C}[S_k]$-module. The induced algebra homomorphism
\[
  \widetilde\rho:\mathbb{C}[S_k]\longrightarrow\text{End}_{\mathbb{C}}(M)
\]
sends each permutation to $W_\sigma$ and has image $A$. These group-algebra
adapters are private to the Lean proof.

By Maschke's theorem, $M$ is semisimple. It is finite-dimensional over
$\mathbb{C}$ and therefore finitely generated over its
$\mathbb{C}[S_k]$-endomorphism ring. Mathlib's finite-module form of Jacobson
density makes the natural map
\[
  \mathbb{C}[S_k]\longrightarrow
    \text{End}_{\text{End}_{\mathbb{C}[S_k]}(M)}(M)
\]
surjective. A $\mathbb{C}[S_k]$-linear endomorphism commutes with every
element of $A$, so any $T\in\text{Comm}(\text{Comm}(A))$ belongs to the
right-hand endomorphism ring. Surjectivity writes $T=\widetilde\rho(r)$,
which puts $T$ in $A$.
} % end revision color

{\chat
\begin{theorem}[Double Commutant Inclusion for Permutation Algebra]\label{thm:double_centralizer_permImage}
  \lean{SchurWeyl.double_centralizer_permImage}\uses{def:permImage, def:permAction, def:TensV}\leanok
  The double centralizer of the span of permutation operators is contained in the span itself:
  \begin{equation}
    \text{Comm}(\text{Comm}(\text{Span}(\mathcal{P}_{d,k}))) \subseteq \text{Span}(\mathcal{P}_{d,k})
  \end{equation}
\end{theorem}
} % end revision color

{\chat
The reverse inclusion $A\subseteq\text{Comm}(\text{Comm}(A))$ holds for
any set of endomorphisms. The Lean theorem above supplies the nontrivial
inclusion used below.

We now assemble Schur--Weyl duality for all natural numbers $d$ and $k$;
no restriction $k\le d$ is needed. If $X$ commutes with every diagonal action,
it commutes with their span. The first fundamental theorem therefore puts
$X$ in the double commutant of $A$, and the preceding inclusion puts it in $A$.
} % end revision color

{\chat
\begin{theorem}[Hard Direction of Schur-Weyl Duality]\label{thm:centralizer_diagImage_le_span_permImage}
  \lean{SchurWeyl.centralizer_diagImage_le_span_permImage}\uses{thm:centralizer_permImage_le_span_diagImage, thm:double_centralizer_permImage, def:diagImage, def:permImage}\leanok
  The centralizer of the diagonal image is contained in the span of the permutation image:
  \begin{equation}
    \text{Comm}(\mathcal{D}_{d,k}) \subseteq \text{Span}(\mathcal{P}_{d,k})
  \end{equation}
\end{theorem}
} % end revision color

\begin{theorem}[Easy Direction of Schur-Weyl Duality]\label{thm:span_permImage_le_centralizer_diagImage}
  \lean{SchurWeyl.span_permImage_le_centralizer_diagImage}\uses{def:permImage, def:diagImage, thm:permImage_subset_centralizer_diagImage}\leanok
  The span of the permutation image is contained in the centralizer of the diagonal image:
  \begin{equation}
    \text{Span}(\mathcal{P}_{d,k}) \subseteq \text{Comm}(\mathcal{D}_{d,k})
  \end{equation}
\end{theorem}

\begin{theorem}[Schur-Weyl Duality]\label{thm:schur_weyl}
  \lean{SchurWeyl.schur_weyl}\uses{thm:centralizer_diagImage_le_span_permImage, thm:span_permImage_le_centralizer_diagImage, def:diagImage, def:permImage}\leanok
  The centralizer of the diagonal image is exactly equal to the span of the permutation image:
  \begin{equation}
    \text{Comm}(\mathcal{D}_{d,k}) = \text{Span}(\mathcal{P}_{d,k})
  \end{equation}
\end{theorem}
